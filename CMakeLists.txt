cmake_minimum_required(VERSION 3.20)
project(grocery_store LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(store src/store.cpp src/repository.cpp src/logging.cpp)
target_include_directories(store PUBLIC include)
add_executable(grocery src/main.cpp)
target_link_libraries(grocery PRIVATE store)
enable_testing()
add_executable(store_tests tests/store_tests.cpp)
target_link_libraries(store_tests PRIVATE store)
foreach(target store grocery store_tests)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endforeach()
add_test(NAME store_tests COMMAND store_tests)
find_package(Python3 3.9 REQUIRED COMPONENTS Interpreter)
if(Python3_FOUND)
  add_test(NAME log_tests COMMAND "${Python3_EXECUTABLE}" -m unittest discover -s tests -p test_logs.py)
  set_tests_properties(log_tests PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
  add_test(NAME application_tests COMMAND "${Python3_EXECUTABLE}" tests/test_application.py "$<TARGET_FILE:grocery>")
  set_tests_properties(application_tests PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
endif()
